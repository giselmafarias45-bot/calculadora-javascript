let opcao = prompt("Digite a opção desejada:")

switch(opcao){

  case "1":
    let n1 = Number(prompt("Digite um número:"))
    let n2 = Number(prompt("Digite outro número:"))
    let resultado = n1 + n2
    console.log(resultado)
    break;

  case "2":
    let n3 = Number(prompt("Digite um número:"))
    let n4 = Number(prompt("Digite outro número:"))
    let resultado2 = n3 - n4
    console.log(resultado2)
    break;

  case "3":
    let n5 = Number(prompt("Digite um número:"))
    let n6 = Number(prompt("Digite outro número:"))
    let resultado3 = n5 * n6
    console.log(resultado3)
    break;

  case "4":
    let n7 = Number(prompt("Digite um número:"))
    let n8 = Number(prompt("Digite outro número:"))
    let resultado4 = n7 / n8
    console.log(resultado4)
    break;

  case "5":
    console.log("Programa encerrado")
    break;

  default:
    console.log("Digite um valor válido")
}
