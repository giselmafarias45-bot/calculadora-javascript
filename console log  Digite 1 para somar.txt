let opcao = ""

while (opcao !== "5") {

  console.log("Digite 1 para somar")
  console.log("Digite 2 para subtrair")
  console.log("Digite 3 para multiplicar")
  console.log("Digite 4 para dividir")
  console.log("Digite 5 para sair")

  opcao = prompt("Digite a opção desejada:")

  switch (opcao) {

    case "1":
      let n1 = Number(prompt("Digite um número:"))
      let n2 = Number(prompt("Digite outro número:"))

      console.log("O resultado da soma é:", n1 + n2)
      break

    case "2":
      let n3 = Number(prompt("Digite um número:"))
      let n4 = Number(prompt("Digite outro número:"))

      console.log("O resultado da subtração é:", n3 - n4)
      break

    case "3":
      let n5 = Number(prompt("Digite um número:"))
      let n6 = Number(prompt("Digite outro número:"))

      console.log("O resultado da multiplicação é:", n5 * n6)
      break

    case "4":
      let n7 = Number(prompt("Digite um número:"))
      let n8 = Number(prompt("Digite outro número:"))

      if (n8 === 0) {
        console.log("Não é possível dividir por zero!")
      } else {
        console.log("O resultado da divisão é:", n7 / n8)
      }
      break

    case "5":
      console.log("Programa encerrado")
      break

    default:
      console.log("Digite um valor válido")
  }
}
